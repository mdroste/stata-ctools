. /*******************************************************************************
>  * sigfigs - Compute number of significant figures of agreement between two values
>  *
>  * Formula: -log10(|a - b| / max(|a|, |b|))
>  *
>  * Returns:
>  *   - 15 (double precision limit) if values are identical
>  *   - 0 if one value is zero and the other is not
>  *   - Number of matching significant figures otherwise
>  *
>  * Examples:
>  *   1.234567 vs 1.234568 → ~6.1 sig figs (differ in 7th digit)
>  *   1000000 vs 1000001   → ~6.0 sig figs
>  *   0.001234 vs 0.001235 → ~3.1 sig figs
>  ******************************************************************************/
. capture program drop sigfigs

. program define sigfigs, rclass
  1.     args val1 val2
  2. 
.     * Handle missing values
.     if missing(`val1') | missing(`val2') {
  3.         return scalar sigfigs = 0
  4.         exit
  5.     }
  6. 
.     * If values are identical (including both zero), return max precision
.     if `val1' == `val2' {
  7.         return scalar sigfigs = 15
  8.         exit
  9.     }
 10. 
.     * If one is zero and other is not, no significant figures agree
.     if (`val1' == 0) | (`val2' == 0) {
 11.         return scalar sigfigs = 0
 12.         exit
 13.     }
 14. 
.     * Compute significant figures of agreement
.     local absdiff = abs(`val1' - `val2')
 15.     local maxabs = max(abs(`val1'), abs(`val2'))
 16.     local reldiff = `absdiff' / `maxabs'
 17. 
.     * Handle edge case where reldiff is 0 (shouldn't happen, but be safe)
.     if `reldiff' == 0 {
 18.         return scalar sigfigs = 15
 19.         exit
 20.     }
 21. 
.     * sigfigs = -log10(reldiff)
.     local sf = -log10(`reldiff')
 22. 
.     * Handle missing result from log10 (extreme values)
.     if missing(`sf') {
 23.         return scalar sigfigs = 0
 24.         exit
 25.     }
 26. 
.     * Clamp to [0, 15] range (double precision limits)
.     if `sf' < 0 {
 27.         local sf = 0
 28.     }
 29.     if `sf' > 15 {
 30.         local sf = 15
 31.     }
 32. 
.     return scalar sigfigs = `sf'
 33. end

. /* Cancellation can leave numerical zero with no meaningful relative digits.
>    Only specialized comparisons supply an operation scale: covariance uses the
>    product of marginal SEs, and 1-ll/ll_0 uses one. This bound is 16 machine eps,
>    not the seven-significant-figure acceptance threshold. Generic comparisons
>    and all values above this roundoff bound retain strict relative precision. */
. capture program drop sigfigs_cancellation

. program define sigfigs_cancellation, rclass
  1.     args val1 val2 scale
  2.     sigfigs `val1' `val2'
  3.     local sf = r(sigfigs)
  4.     local bound = 16*c(epsdouble)*`scale'
  5.     if !missing(`val1', `val2', `bound') & `bound' > 0 {
  6.         if max(abs(`val1'),abs(`val2')) <= `bound' & abs(`val1'-`val2') <= `bound' {
  7.             local sf = 15
  8.         }
  9.     }
 10.     return scalar sigfigs = `sf'
 11. end

  1.     args var1 var2 min_sigfigs testname
  2.

  3.

  4.     quietly {
  5.         * Handle identical values (including both zero)

 10.         replace `sf' = 0 if missing(`sf')
 11.     }
 12.

 13.     local ndiff = r(N)
 14.

 15.         test_pass "`testname'"
 16.     }
 17.     else {
 18.         quietly summarize `sf'
 19.         local min_sf : display %5.1f r(min)

. /*******************************************************************************
>  * sigfigs - Compute number of significant figures of agreement between two values
>  *
>  * Formula: -log10(|a - b| / max(|a|, |b|))
>  *
>  * Returns:
>  *   - 15 (double precision limit) if values are identical
>  *   - 0 if one value is zero and the other is not
>  *   - Number of matching significant figures otherwise
>  *
>  * Examples:
>  *   1.234567 vs 1.234568 → ~6.1 sig figs (differ in 7th digit)
>  *   1000000 vs 1000001   → ~6.0 sig figs
>  *   0.001234 vs 0.001235 → ~3.1 sig figs
>  ******************************************************************************/
. capture program drop sigfigs

. program define sigfigs, rclass
  1.     args val1 val2
  2. 
.     * Handle missing values
.     if missing(`val1') | missing(`val2') {
  3.         return scalar sigfigs = 0
  4.         exit
  5.     }
  6. 
.     * If values are identical (including both zero), return max precision
.     if `val1' == `val2' {
  7.         return scalar sigfigs = 15
  8.         exit
  9.     }
 10. 
.     * If one is zero and other is not, no significant figures agree
.     if (`val1' == 0) | (`val2' == 0) {
 11.         return scalar sigfigs = 0
 12.         exit
 13.     }
 14. 
.     * Compute significant figures of agreement
.     local absdiff = abs(`val1' - `val2')
 15.     local maxabs = max(abs(`val1'), abs(`val2'))
 16.     local reldiff = `absdiff' / `maxabs'
 17. 
.     * Handle edge case where reldiff is 0 (shouldn't happen, but be safe)
.     if `reldiff' == 0 {
 18.         return scalar sigfigs = 15
 19.         exit
 20.     }
 21. 
.     * sigfigs = -log10(reldiff)
.     local sf = -log10(`reldiff')
 22. 
.     * Handle missing result from log10 (extreme values)
.     if missing(`sf') {
 23.         return scalar sigfigs = 0
 24.         exit
 25.     }
 26. 
.     * Clamp to [0, 15] range (double precision limits)
.     if `sf' < 0 {
 27.         local sf = 0
 28.     }
 29.     if `sf' > 15 {
 30.         local sf = 15
 31.     }
 32. 
.     return scalar sigfigs = `sf'
 33. end

. 
. /* Cancellation can leave numerical zero with no meaningful relative digits.
>    Only specialized comparisons supply an operation scale: covariance uses the
>    product of marginal SEs, and 1-ll/ll_0 uses one. This bound is 16 machine eps,
>    not the seven-significant-figure acceptance threshold. Generic comparisons
>    and all values above this roundoff bound retain strict relative precision. */
. capture program drop sigfigs_cancellation

. program define sigfigs_cancellation, rclass
  1.     args val1 val2 scale
  2.     sigfigs `val1' `val2'
  3.     local sf = r(sigfigs)
  4.     local bound = 16*c(epsdouble)*`scale'
  5.     if !missing(`val1', `val2', `bound') & `bound' > 0 {
  6.         if max(abs(`val1'),abs(`val2')) <= `bound' & abs(`val1'-`val2') <= `bound' {
  7.             local sf = 15
  8.         }
  9.     }
 10.     return scalar sigfigs = `sf'
 11. end

. /*******************************************************************************
>  * sigfigs - Compute number of significant figures of agreement between two values
>  *
>  * Formula: -log10(|a - b| / max(|a|, |b|))
>  *
>  * Returns:
>  *   - 15 (double precision limit) if values are identical
>  *   - 0 if one value is zero and the other is not
>  *   - Number of matching significant figures otherwise
>  *
>  * Examples:
>  *   1.234567 vs 1.234568 → ~6.1 sig figs (differ in 7th digit)
>  *   1000000 vs 1000001   → ~6.0 sig figs
>  *   0.001234 vs 0.001235 → ~3.1 sig figs
>  ******************************************************************************/
. capture program drop sigfigs

. program define sigfigs, rclass
  1.     args val1 val2
  2. 
.     * Handle missing values
.     if missing(`val1') | missing(`val2') {
  3.         return scalar sigfigs = 0
  4.         exit
  5.     }
  6. 
.     * If values are identical (including both zero), return max precision
.     if `val1' == `val2' {
  7.         return scalar sigfigs = 15
  8.         exit
  9.     }
 10. 
.     * If one is zero and other is not, no significant figures agree
.     if (`val1' == 0) | (`val2' == 0) {
 11.         return scalar sigfigs = 0
 12.         exit
 13.     }
 14. 
.     * Compute significant figures of agreement
.     local absdiff = abs(`val1' - `val2')
 15.     local maxabs = max(abs(`val1'), abs(`val2'))
 16.     local reldiff = `absdiff' / `maxabs'
 17. 
.     * Handle edge case where reldiff is 0 (shouldn't happen, but be safe)
.     if `reldiff' == 0 {
 18.         return scalar sigfigs = 15
 19.         exit
 20.     }
 21. 
.     * sigfigs = -log10(reldiff)
.     local sf = -log10(`reldiff')
 22. 
.     * Handle missing result from log10 (extreme values)
.     if missing(`sf') {
 23.         return scalar sigfigs = 0
 24.         exit
 25.     }
 26. 
.     * Clamp to [0, 15] range (double precision limits)
.     if `sf' < 0 {
 27.         local sf = 0
 28.     }
 29.     if `sf' > 15 {
 30.         local sf = 15
 31.     }
 32. 
.     return scalar sigfigs = `sf'
 33. end

. 
. /* Cancellation can leave numerical zero with no meaningful relative digits.
>    Only specialized comparisons supply an operation scale: covariance uses the
>    product of marginal SEs, and 1-ll/ll_0 uses one. This bound is 16 machine eps,
>    not the seven-significant-figure acceptance threshold. Generic comparisons
>    and all values above this roundoff bound retain strict relative precision. */
. capture program drop sigfigs_cancellation

. program define sigfigs_cancellation, rclass
  1.     args val1 val2 scale
  2.     sigfigs `val1' `val2'
  3.     local sf = r(sigfigs)
  4.     local bound = 16*c(epsdouble)*`scale'
  5.     if !missing(`val1', `val2', `bound') & `bound' > 0 {
  6.         if max(abs(`val1'),abs(`val2')) <= `bound' & abs(`val1'-`val2') <= `bound' {
  7.             local sf = 15
  8.         }
  9.     }
 10.     return scalar sigfigs = `sf'
 11. end
